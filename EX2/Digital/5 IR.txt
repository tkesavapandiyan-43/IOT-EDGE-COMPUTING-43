const int irPin = 2;
const int ledPin = 13;

void setup() {
  pinMode(irPin, INPUT);
  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int irState = digitalRead(irPin);

  if (irState == LOW) {      // Object detected
    digitalWrite(ledPin, HIGH);
    Serial.println("Object Detected");
  } else {                   // No object
    digitalWrite(ledPin, LOW);
    Serial.println("No Object");
  }

  delay(100);
}
