EXPERIMENT: Web User Interface LED ON/OFF using Ethernet Shield

#include <SPI.h>
#include <Ethernet.h>

// Ethernet Network Configuration
byte mac[] = {
  0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED
};

IPAddress ip(192, 168, 1, 20);
IPAddress gateway(192, 168, 1, 1);
IPAddress subnet(255, 255, 255, 0);

EthernetServer server(80);

#define LED 7

void setup() {
  pinMode(LED, OUTPUT);

  Serial.begin(9600);

  Ethernet.begin(mac, ip, gateway, gateway, subnet);
  server.begin();

  Serial.print("Server started at: ");
  Serial.println(Ethernet.localIP());
}

void loop() {
  EthernetClient client = server.available();

  if (client) {
    boolean currentLineIsBlank = true;
    String request = "";

    while (client.connected()) {
      if (client.available()) {
        char c = client.read();
        request += c;

        if (c == '\n' && currentLineIsBlank) {

          // Check the requested URL
          if (request.indexOf("GET /LEDON") >= 0) {
            digitalWrite(LED, HIGH);
          }

          if (request.indexOf("GET /LEDOFF") >= 0) {
            digitalWrite(LED, LOW);
          }

          // HTTP response
          client.println("HTTP/1.1 200 OK");
          client.println("Content-Type: text/html");
          client.println("Connection: close");
          client.println();

          client.println("<!DOCTYPE HTML>");
          client.println("<html>");
          client.println("<head>");
          client.println("<title>Arduino LED Control</title>");
          client.println("</head>");
          client.println("<body>");
          client.println("<h1>Arduino Ethernet LED Control</h1>");

          client.println("<p>LED Status: ");

          if (digitalRead(LED) == HIGH) {
            client.println("ON</p>");
          }
          else {
            client.println("OFF</p>");
          }

          client.println("<p><a href="/LEDON">");
          client.println("<button>LED ON</button>");
          client.println("</a></p>");

          client.println("<p><a href="/LEDOFF">");
          client.println("<button>LED OFF</button>");
          client.println("</a></p>");

          client.println("</body>");
          client.println("</html>");

          break;
        }

        if (c == '\n') {
          currentLineIsBlank = true;
        }
        else if (c != '\r') {
          currentLineIsBlank = false;
        }
      }
    }

    delay(1);
    client.stop();
  }
}
