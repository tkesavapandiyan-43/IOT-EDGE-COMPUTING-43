#include <SoftwareSerial.h>

SoftwareSerial BT(10, 11);  // RX, TX

#define LED 7

void setup() {
  pinMode(LED, OUTPUT);

  Serial.begin(9600);
  BT.begin(9600);

  Serial.println("Bluetooth LED Control Ready");
  BT.println("Send 1 = LED ON");
  BT.println("Send 0 = LED OFF");
}

void loop() {

  if (BT.available()) {

    char data = BT.read();

    if (data == '1') {
      digitalWrite(LED, HIGH);
      BT.println("LED ON");
    }

    else if (data == '0') {
      digitalWrite(LED, LOW);
      BT.println("LED OFF");
    }
  }
}
