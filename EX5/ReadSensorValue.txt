#include <SoftwareSerial.h>

SoftwareSerial BT(10, 11);  // RX, TX

#define LDR A0

void setup() {
  Serial.begin(9600);
  BT.begin(9600);

  Serial.println("Bluetooth LDR Sensor Ready");
  BT.println("LDR Sensor Ready");
}

void loop() {

  int sensorValue = analogRead(LDR);

  Serial.print("LDR Value: ");
  Serial.println(sensorValue);

  BT.print("LDR Value: ");
  BT.println(sensorValue);

  delay(1000);
}
